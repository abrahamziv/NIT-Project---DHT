\section{Problem Formulation}
Consider binary hypotheses $H_1$ and $H_2$, and let
$H^\star\in\{1,2\}$ index the true one, with priors $P(H^\star=1)=p$ and
$P(H^\star=2)=1-p$.  Sensor $i\in\{1,\ldots,N\}$
observes $y^i\in\mathcal{Y}$ and applies an encoder
$\gamma^i:\mathcal{Y}\to\mathcal{U}$.  Its message
$u^i=\gamma^i(y^i)$ is available to a fusion center (FC), which uses
$\gamma^0:\mathcal{U}^N\to\{1,2\}$ to produce the final decision
$u^0=\gamma^0(u^{1:N})$.

Under uniform error cost the Bayes risk is the probability of error,
\begin{equation}
 J^N(\gamma^{0:N}) \triangleq
 \mathbb{P}^{\gamma^{0:N}}\big(u^0\neq H^\star\big) = P_e
 \label{eq:bayes-risk}
\end{equation}
We use the positive normalized error exponent
\begin{equation}
 J^N_{EE}(\gamma^{0:N}) \triangleq -\frac{1}{N}\log J^N(\gamma^{0:N}),
 \label{eq:error-exponent}
\end{equation}
which is maximized when the error probability decays rapidly.

\subsection{Assumptions}
The analysis uses the following assumptions.
\begin{enumerate}
 \item The observations $y^1,\ldots,y^N$ are independent and
 identically distributed conditioned on the true hypothesis.
 \item Under both hypotheses, the observation laws have densities
 $f_{y|H}(y|H_j)$ with respect to a common reference
 measure.
 \item All sensors have the same finite message alphabet $\mathcal{U}$.
 \item The likelihood ratio
 \begin{equation*}
  L(y):=\frac{f_{y|H}(y|H_2)}
  {f_{y|H}(y|H_1)}
 \end{equation*}
 is regular enough that threshold encoders are well defined; in
 particular, its derivative is continuous and nonsingular almost
 surely on the observation space.
 \item The FC knows the sensor encoders, including any realized common
 randomization, as well as the received messages.
\end{enumerate}

Under these conditions, the likelihood ratio is a sufficient statistic
at each sensor.  The results of \cite{Sanjari2025} guarantee an optimal
finite-population solution with threshold encoders and a maximum a
posteriori (MAP) decoder.  They also show that identical independent
sensor policies are optimal in the infinite-population problem and
asymptotically optimal for large finite populations.

\subsection{MAP fusion rule}
For fixed deterministic encoders, define
\begin{equation}
 \Delta_N \triangleq \frac{1}{N}\sum_{i=1}^{N}
 \log\frac{P^{\gamma^i}(u^i|H_1)}
 {P^{\gamma^i}(u^i|H_2)}
 \label{eq:delta}
\end{equation}
The MAP rule is
\begin{equation}
 \gamma^{0\star}(u^{1:N})=
 \begin{cases}
  H_1 & \text{if } \Delta_N\ge t\\
  H_2 & \text{if } \Delta_N<t
 \end{cases}
 \quad
 t=\frac{1}{N}\log\frac{1-p}{p}
 \label{eq:map}
\end{equation}

\section{Error-Exponent Characterization}
For a sensor policy $\gamma^i$, define the policy-dependent Chernoff
coefficient
\begin{equation}
 M(\alpha,\gamma^i) \triangleq \sum_{u\in\mathcal{U}}
 P^{\gamma^i}(u|H_2)^{1-\alpha}
 P^{\gamma^i}(u|H_1)^{\alpha},
 \quad 0\le\alpha\le1.
 \label{eq:chernoff-coefficient}
\end{equation}
Applying Chernoff bounds to the two MAP error events gives
\begin{equation}
 P_e\le 2p^{\alpha}(1-p)^{1-\alpha}
 \prod_{i=1}^{N}M(\alpha,\gamma^i)
 \label{eq:pe-bound}
\end{equation}
When every sensor uses the same policy $\gamma$, taking
$-\tfrac{1}{N}\log(\cdot)$ of \eqref{eq:pe-bound} and letting
$N\to\infty$ leaves the Chernoff information between the two induced
message distributions,
\begin{equation}
 \liminf_{N\to\infty}J^N_{EE}(\gamma^{0:N})
 \ge -\min_{\alpha\in[0,1]}\log M(\alpha,\gamma)
 \label{eq:identical-exponent}
\end{equation}
The structural result in \cite{Sanjari2025} therefore reduces the
large-network design to selecting one encoder that maximizes this
exponent, subject here to a communication-rate constraint.

\section{Rate-Aware $R$-Bit Threshold Policies}
Assume
\begin{equation}
 y^i|H_1\sim\mathcal{N}(0,\sigma^2),\qquad
 y^i|H_2\sim\mathcal{N}(\mu,\sigma^2),
 \label{eq:gaussian-model}
\end{equation}
with $\mu>0$.  Then
\begin{equation}
 \log L(y)=\frac{\mu y}{\sigma^2}-\frac{\mu^2}{2\sigma^2}
 \label{eq:gaussian-llr}
\end{equation}
We generalize the always-transmit policy and the confidence-based silent
policy into a single family of threshold policies, indexed by the number of
bits $R$, a bin width $\Delta>0$, and a silence half-width $\delta\ge0$.

\subsection{The $R$-bit ladder}
Let $L\triangleq2^R$ and $\mathcal{U}=\{u^1,\dots,u^L,u^S\}$, where $u^S$
denotes silence.  For $\delta\ge0$, define the thresholds
\begin{equation}
 t_j(\delta)\triangleq
 \begin{cases}
  \dfrac{\mu}{2}-\delta-\Big(\dfrac{L}{2}-j\Big)\Delta,
    & j=1,\dots,\dfrac{L}{2}\\[2mm]
  \dfrac{\mu}{2}+\delta+\Big(j-\dfrac{L}{2}-1\Big)\Delta,
    & j=\dfrac{L}{2}+1,\dots,L
 \end{cases}
 \label{eq:general-thresholds}
\end{equation}
with $t_0\triangleq-\infty$ and $t_{L+1}\triangleq+\infty$, so the
thresholds cut the line into $L+1$ intervals $(t_{j-1},t_j)$,
$j=1,\dots,L+1$.  The sensor sends $u^j$ from the $j$-th interval for
$j\le L/2$, stays silent on the central one, $j=L/2+1$, and sends
$u^{j-1}$ for $j\ge L/2+2$.

At $\delta=0$ the two central thresholds coincide at $\mu/2$: the silence
interval degenerates to a point, $u^S$ is never sent, and what remains is
an $L$-region always-transmit policy.  We write $\gamma^A$ for the policy
at $\delta=0$, and $\gamma^S$ for the policy at $\delta>0$.

\subsection{Symbol probabilities}
Standardizing by $\sigma$, write
\begin{equation*}
 a_j\triangleq\frac{t_j(\delta)}{\sigma},\qquad
 s\triangleq\frac{\mu}{\sigma},
\end{equation*}
with $a_0\triangleq-\infty$ and $a_{L+1}\triangleq+\infty$.  Since
$\log L(y)$ in \eqref{eq:gaussian-llr} is increasing in $y$, every
decision region is an interval and its probability is a difference of
$Q$-functions,
\begin{align*}
 P(t_{j-1}<y<t_j\mid H_1)&=\Qfun(a_{j-1})-\Qfun(a_j),\\
 P(t_{j-1}<y<t_j\mid H_2)&=\Qfun(a_{j-1}-s)-\Qfun(a_j-s)
\end{align*}
Every quantity below is written in terms of $a_j$ and $s$, and applies to
$\gamma^A$ and $\gamma^S$ alike.

\subsection{Symmetry and the closed-form bound}
The thresholds \eqref{eq:general-thresholds} mirror about $\mu/2$, that is
$t_{L+1-j}(\delta)=\mu-t_j(\delta)$, and the two densities mirror with them,
$f_{y|H}(y|H_1)=f_{y|H}(\mu-y|H_2)$.  The message probabilities are
therefore exchanged by the mirror, $P(u^m|H_1)=P(u^{L+1-m}|H_2)$, and the
silence probability is hypothesis-independent,
\begin{align}
 P(u^S|H_1)=P(u^S|H_2)&\triangleq P(u^S)\nonumber\\
 &=\Qfun\Big(\frac{s}{2}-\frac{\delta}{\sigma}\Big)
 -\Qfun\Big(\frac{s}{2}+\frac{\delta}{\sigma}\Big)
 \label{eq:silence-prob}
\end{align}
Generalizing \eqref{eq:chernoff-coefficient} to the $L+1$ symbols, the
silence term collapses for every $\alpha$ by \eqref{eq:silence-prob}:
\begin{equation*}
 M(\alpha;R,\Delta,\delta)\triangleq P(u^S)
 +\sum_{m=1}^{L}P(u^m|H_2)^{1-\alpha}P(u^m|H_1)^{\alpha}
\end{equation*}
The mirror relation gives $M(1-\alpha;\cdot)=M(\alpha;\cdot)$, and each
summand is convex in $\alpha$, so $M$ is convex on $[0,1]$ and symmetric
about $\alpha=1/2$: its minimum sits there.  With
\eqref{eq:identical-exponent}, this gives the bound in closed form,
\begin{align}
 C(R,\Delta,\delta)&\triangleq-\log M(0.5;R,\Delta,\delta)\nonumber\\
 &=-\log\Big[P(u^S)+\sum_{m=1}^{L}\sqrt{P(u^m|H_1)P(u^m|H_2)}\Big]
 \label{eq:general-closed-form}
\end{align}
\begin{equation}
 \liminf_{N\to\infty}J^N_{EE}(\gamma^{0:N})\ge C(R,\Delta,\delta)
 \label{eq:general-closed-form-bound}
\end{equation}
At $\delta=0$ we have $P(u^S)=0$ and \eqref{eq:general-closed-form} reduces
to the $L$-region always-transmit bound $C_A\triangleq C(R,\Delta,0)$; for
$\delta>0$ we write $C_S\triangleq C(R,\Delta,\delta)$.

\subsection{The unquantized ceiling}
Before evaluating the bound it is worth fixing what it is bounded by.
Suppose the sensor forwards $y^i$ itself, at infinite precision.  Under the
Gaussian model \eqref{eq:gaussian-model} its Chernoff coefficient is
$\exp\big(-\alpha(1-\alpha)s^2/2\big)$, again symmetric about
$\alpha=1/2$ and convex, so its minimum sits there as in
\eqref{eq:general-closed-form}, giving
\begin{align}
 C_\infty&\triangleq-\min_{\alpha\in[0,1]}\log\int
 f_{y|H}^{1-\alpha}(y|H_2)\,f_{y|H}^{\alpha}(y|H_1)\,dy\nonumber\\
 &=\frac{\mu^2}{8\sigma^2}=\frac{s^2}{8}
 \label{eq:ceiling}
\end{align}
Since $H^\star\to y^i\to u^i$ is Markov for any policy, the data processing
inequality gives $I(H^\star;u^i)\le I(H^\star;y^i)$: quantization can
discard evidence about the hypothesis, never create it.  Hence
\begin{equation}
 C(R,\Delta,\delta)\le C_\infty\qquad\text{for every }R,\Delta,\delta,
 \label{eq:ceiling-bound}
\end{equation}
and $C_\infty$ is the natural yardstick against which every finite-rate
policy is measured.  Note that $C_\infty$ depends on the model only through
$s=\mu/\sigma$: the whole problem is how many standard deviations apart the
two hypotheses sit.

\subsection{Transmitted rate and the design problem}
The parameter $R$ is the rate $\gamma^A$ pays, but not the rate $\gamma^S$
pays: a silent sensor sends nothing.  Using \eqref{eq:silence-prob}, the
expected transmitted rate is
\begin{equation}
 \bar R\triangleq R\big(1-P(u^S)\big),
 \label{eq:mean-rate}
\end{equation}
so $\bar R=R$ for $\gamma^A$ and $\bar R<R$ for $\gamma^S$.  Comparisons
made at equal $R$ therefore charge $\gamma^S$ for bandwidth it never uses.

At a given rate the design problem is to place the ladder,
\begin{equation}
 (\Delta^\star,\delta^\star)
 =\arg\max_{\Delta>0,\ \delta\ge0}C(R,\Delta,\delta),
 \label{eq:design-problem}
\end{equation}
with $\gamma^A$ obtained by pinning $\delta=0$ and optimizing over $\Delta$
alone.  For $R>1$ this admits no closed form, and we solve it numerically.

\section{Numerical Results}
\label{sec:numerical}
We evaluate the ladder for $R=1,\dots,5$ at $N=3000$ sensors
and $p=\tfrac12$, at three operating points
$\mathrm{SNR}\in\{-5,0,+5\}$~dB.  At each point $(\Delta^\star,
\delta^\star)$ solves \eqref{eq:design-problem}.  The fusion center computes
$J^N$ exactly, from which $J^N_{EE}$ follows by \eqref{eq:error-exponent}.

\begin{figure}[t]
 \centering
 \includegraphics[width=\linewidth]{Figs/fig_icassp_JEE_vs_R_N3000_snrp0.png}
 \caption{Normalized error exponent $J^N_{EE}$ against $R$ at the optimal
 $(\Delta^\star,\delta^\star)$, for $N=3000$ and $\mathrm{SNR}=0$~dB, with
 the closed-form bounds of \eqref{eq:general-closed-form} (dotted) and the
 unquantized ceiling $C_\infty=0.125$ of \eqref{eq:ceiling} (dashed).}
 \label{fig:jee-vs-R}
\end{figure}

Figure~\ref{fig:jee-vs-R} shows the $0$~dB panel, and
Figures~\ref{fig:jee-m5} and~\ref{fig:jee-p5} the other two operating
points; one shape repeats at all three.  $J^N_{EE}$ rises monotonically in
$R$ and approaches the unquantized ceiling $C_\infty$ of
\eqref{eq:ceiling}.  It sits slightly above its own bound throughout -- by
$0.0012$ to $0.0016$ across the three panels at $N=3000$, essentially flat
in $R$ -- since $C$ bounds the $N\to\infty$ limit while the finite-$N$
exponent still carries an $O(1/N)$ prefactor.  The same offset carries it a
little past $C_\infty$ itself at high $R$, most visibly at $-5$~dB where the
ceiling is smallest: \eqref{eq:ceiling-bound} constrains the limit, not the
finite-$N$ exponent.

Saturation is fast: a single bit already recovers $63\%$ of $C_\infty$ under
$\gamma^A$ and $81\%$ under $\gamma^S$, and by $R=3$ both policies sit
within a few percent of the ceiling.  The entire design question therefore
lives at $R\le3$; past that the extra bits buy nothing.

The silent policy dominates the always-transmit one at every $R$, as $C_S>C_A$
predicts, but the whole of that advantage is concentrated at low rate.  It
is largest at $R=1$ and by $R=4$ the two curves are indistinguishable.  The
optimized geometry explains why.  Table~\ref{tab:opt-params} lists
$(\Delta^\star,\delta^\star)$ against $R$: the cell width $\Delta^\star$
decays geometrically, settling toward a factor of about $0.55$ per added
bit, while the ratio $\delta^\star/\Delta^\star$ stays close to $0.4$
throughout.  The silence cell holds an essentially fixed fraction of a cell,
so it shrinks with the cell; by \eqref{eq:silence-prob} the silence
probability vanishes with it, and $\gamma^S$ degenerates continuously into
$\gamma^A$.  The two effects visible at the knee of
Figure~\ref{fig:jee-vs-R} -- silence ceasing to help, and quantization
ceasing to cost -- are the same phenomenon, $\Delta^\star\to0$, seen from
two sides.

Across the three operating points only the vertical scale changes.  The
ceiling moves from $C_\infty\approx0.0395$ at $-5$~dB to $0.125$ at
$0$~dB to $0.3953$ at $+5$~dB, an order of magnitude, while the shape of
both curves is preserved: normalized by $C_\infty$ the bound curves agree
to within $0.7$ percentage points, and $(\Delta^\star,\delta^\star)$ in
units of $\sigma$ barely move.  This is \eqref{eq:ceiling} showing through
-- the model enters only through $s$, which $C_\infty$ absorbs.  SNR sets
the achievable exponent; $R$ sets what fraction of it is achieved.

Both policies are charged the nominal $R$, which by \eqref{eq:mean-rate}
overstates what $\gamma^S$ sends: at $R=1$ and $0$~dB it transmits
$\bar R\approx0.59$ bits per sensor, below the one bit $\gamma^A$ must
spend.

\section{Conclusion}
We combined the optimal-policy results for large sensor populations with a
rate-aware encoder, generalizing the one-bit always-transmit and silent
policies into a single $R$-bit threshold family whose error exponent is
available in closed form.  Measured against the unquantized ceiling
$C_\infty=s^2/8$, the family saturates quickly: one bit per sensor already
buys $63\%$ of $C_\infty$ without silence and $81\%$ with it, three bits
come within a few percent, and beyond that the extra rate is wasted.
Selective silence is therefore worth the most at exactly the rate where rate
is scarcest -- at $R=1$ it raises the exponent while transmitting
$\bar R\approx0.59$ bits per sensor, where the always-transmit policy must
spend a full bit -- and its advantage vanishes as the optimal cell width
shrinks and the silence cell closes with it.
