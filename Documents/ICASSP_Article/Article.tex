\section{Problem Formulation}
Consider binary hypotheses $H_1$ and $H_2$ with priors
$P(H_1)=p$ and $P(H_2)=1-p$.  Sensor $i\in\{1,\ldots,N\}$
observes $y^i\in\mathcal{Y}$ and applies an encoder
$\gamma^i:\mathcal{Y}\to\mathcal{U}$.  Its message
$u^i=\gamma^i(y^i)$ is available to a fusion center (FC), which uses
$\gamma^0:\mathcal{U}^N\to\{1,2\}$ to produce the final decision
$u^0=\gamma^0(u^{1:N})$.

Under uniform error cost,
\begin{equation}
 c(H_j,u^0)=
 \begin{cases}
  1 & \text{if } u^0 \neq j\\
  0 & \text{if } u^0 = j
 \end{cases}
 \label{eq:cost}
\end{equation}
\begin{equation}
 J^N(\gamma^{0:N}) \triangleq \mathbb{E}^{\gamma^{0:N}}[c(H^\star,u^0)].
 \label{eq:bayes-risk}
\end{equation}
Thus, $J^N=P_e$ is the Bayes probability of error.  We use the
positive normalized error exponent
\begin{equation}
 J^N_{EE}(\gamma^{0:N}) \triangleq -\frac{1}{N}\log J^N(\gamma^{0:N}),
 \label{eq:error-exponent}
\end{equation}
which is maximized when the error probability decays rapidly.

\subsection{Assumptions}
The analysis uses the following assumptions.
\begin{enumerate}
 \item The observations $y^1,\ldots,y^N$ are independent and
 identically distributed conditioned on the true hypothesis.
 \item Under both hypotheses, the observation laws have densities
 $f_{y|H}(y|H_j)$ with respect to a common reference
 measure.
 \item All sensors have the same finite message alphabet $\mathcal{U}$.
 \item The likelihood ratio
 \begin{equation}
  L(y):=\frac{f_{y|H}(y|H_2)}
  {f_{y|H}(y|H_1)}
  \label{eq:lr}
 \end{equation}
 is regular enough that threshold encoders are well defined; in
 particular, its derivative is continuous and nonsingular almost
 surely on the observation space.
 \item The FC knows the sensor encoders, including any realized common
 randomization, as well as the received messages.
\end{enumerate}

Under these conditions, the likelihood ratio is a sufficient statistic
at each sensor.  The results of \cite{Sanjari2025} guarantee an optimal
finite-population solution with threshold encoders and a maximum a
posteriori (MAP) decoder.  They also show that identical independent
sensor policies are optimal in the infinite-population problem and
asymptotically optimal for large finite populations.

\subsection{MAP fusion rule}
For fixed deterministic encoders, define
\begin{equation}
 \Delta_N(u^{1:N};\gamma^{1:N}) \triangleq \frac{1}{N}\sum_{i=1}^{N}
 \log\frac{P^{\gamma^i}(u^i|H_1)}
 {P^{\gamma^i}(u^i|H_2)}
 \label{eq:delta}
\end{equation}
The MAP rule is
\begin{equation}
 \gamma^{0\star}(u^{1:N})=
 \begin{cases}
  H_1 & \text{if } \Delta_N(u^{1:N};\gamma^{1:N})\ge t\\
  H_2 & \text{if } \Delta_N(u^{1:N};\gamma^{1:N})<t
 \end{cases}
 \quad
 t=\frac{1}{N}\log\frac{1-p}{p}
 \label{eq:map}
\end{equation}

\section{Error-Exponent Characterization}
For a sensor policy $\gamma^i$, define the policy-dependent Chernoff
coefficient
\begin{equation}
 M(\alpha,\gamma^i) \triangleq \sum_{u\in\mathcal{U}}
 P^{\gamma^i}(u|H_2)^{1-\alpha}
 P^{\gamma^i}(u|H_1)^{\alpha},
 \quad 0\le\alpha\le1.
 \label{eq:chernoff-coefficient}
\end{equation}
Applying Chernoff bounds to the two MAP error events gives
\begin{equation}
 P_e\le 2p^{\alpha}(1-p)^{1-\alpha}
 \prod_{i=1}^{N}M(\alpha,\gamma^i)
 \label{eq:pe-bound}
\end{equation}
Suppose that the sensors use $K$ distinct policies and that a fraction
$c_k$ uses policy $\gamma^k$, where $c_k\ge0$ and
$\sum_{k=1}^{K}c_k=1$.  From \eqref{eq:pe-bound},
\begin{equation}
 \liminf_{N\to\infty}J^N_{EE}(\gamma^{0:N})
 \ge -\min_{\alpha\in[0,1]}
 \sum_{k=1}^{K}c_k\log M(\alpha,\gamma^k)
 \label{eq:mixed-exponent}
\end{equation}
For an identical policy $\gamma$, this becomes the Chernoff information
between the two induced message distributions:
\begin{equation}
 C_\gamma=-\min_{\alpha\in[0,1]}\log
 \left(\sum_{u\in\mathcal{U}}
 P^{\gamma}(u|H_2)^{1-\alpha}
 P^{\gamma}(u|H_1)^{\alpha}\right)
 \label{eq:chernoff-information}
\end{equation}
The structural result in \cite{Sanjari2025} therefore reduces the
large-network design to selecting one encoder that maximizes
$C_\gamma$, subject here to a communication-rate constraint.

\section{Rate-Aware Gaussian Policies}
Assume
\begin{equation}
 y^i|H_1\sim\mathcal{N}(0,\sigma^2),\qquad
 y^i|H_2\sim\mathcal{N}(\mu,\sigma^2),
 \label{eq:gaussian-model}
\end{equation}
with $\mu>0$.  Then
\begin{equation}
 \log L(y)=\frac{\mu y}{\sigma^2}-\frac{\mu^2}{2\sigma^2}
 \label{eq:gaussian-llr}
\end{equation}
We compare an always-transmit binary policy with a confidence-based policy that can remain silent, allocating one bit per sensor.
\subsection{Always-transmit policy}
For the symmetric threshold $\lambda_V=\mu/2$, let $u^1$ be
sent below the threshold and $u^2$ otherwise.  With
$q=\Qfun(\mu/(2\sigma))$, the induced probabilities are
\begin{align}
 P(u^1|H_1)&=P(u^2|H_2)=1-q,\nonumber\\
 P(u^2|H_1)&=P(u^1|H_2)=q
 \label{eq:vanilla-probabilities}
\end{align}
The Chernoff coefficient is symmetric about $\alpha=1/2$, and hence
\begin{equation}
 C_V=-\log\left(2\sqrt{q(1-q)}\right)
 \label{eq:vanilla-exponent}
\end{equation}
Every sensor transmits one bit, so the total rate is
$R_V=N$ bits per observation interval.

\subsection{Silent policy}
Introduce a no-transmission symbol $u^S$ and symmetric
thresholds
\begin{equation}
 \lambda_1=\frac{\mu}{2}-\delta,\qquad
 \lambda_2=\frac{\mu}{2}+\delta,
 \quad \delta>0.
 \label{eq:silent-thresholds}
\end{equation}
The sensor sends $u^1$ for $y<\lambda_1$, remains silent for
$\lambda_1\le y\le\lambda_2$, and sends $u^2$ for
$y>\lambda_2$.  Define
\begin{equation}
 q_+=\Qfun\!\left(\frac{\mu/2+\delta}{\sigma}\right),\qquad
 q_-=\Qfun\!\left(\frac{\mu/2-\delta}{\sigma}\right)
 \label{eq:qpm}
\end{equation}
The message probabilities are
\begin{align}
 (P(u^1|H_1),P(u^S|H_1),P(u^2|H_1))
   &=(1-q_-,q_--q_+,q_+),\nonumber\\
 (P(u^1|H_2),P(u^S|H_2),P(u^2|H_2))
   &=(q_+,q_--q_+,1-q_-)
 \label{eq:silent-probabilities}
\end{align}
Again, symmetry gives the minimizing Chernoff parameter
$\alpha=1/2$, so
\begin{equation}
 C_S(\delta)=-\log\left(
 q_--q_+ +2\sqrt{(1-q_-)q_+}\right)
 \label{eq:silent-exponent}
\end{equation}
The probability of transmission is
$\rho(\delta)=1-q_-+q_+$ under either hypothesis.  If an active sensor
sends one bit and silence is detected at no bit cost, then
\begin{equation}
 R_S(\delta)=N\rho(\delta)
 \label{eq:silent-rate}
\end{equation}
is the expected total rate.  Equations \eqref{eq:silent-exponent} and
\eqref{eq:silent-rate} expose the detection--communication tradeoff;
$\delta$ can be selected by
\begin{equation}
 \max_{\delta>0}\ C_S(\delta)
 \quad\text{subject to}\quad R_S(\delta)\le R_{\max}
 \label{eq:rate-optimization}
\end{equation}

\section{Optimal Silence Threshold}
\label{sec:optimal-delta}
At $\alpha=1/2$, maximizing $C_S(\delta)$ in \eqref{eq:silent-exponent}
is equivalent to minimizing
\begin{equation}
 M_S(\delta) = (q_- - q_+) + 2\sqrt{(1-q_-)q_+}
 \label{eq:ms-of-delta}
\end{equation}
Writing $\phi_\pm \triangleq \phi\big((\mu/2\pm\delta)/\sigma\big)$ for
the standard normal density, $A\triangleq\sqrt{q_+}$, and
$B\triangleq\sqrt{1-q_-}$, differentiating \eqref{eq:ms-of-delta} with
respect to $\delta$ gives
\begin{align}
 \sigma\frac{dM_S}{d\delta}
 = \frac{(B-A)(\phi_-A-\phi_+B)}{AB}
 \label{eq:dms}
\end{align}
Since $B>A$ for all $\delta>0$ (because
$B^2=\Qfun((\delta-\mu/2)/\sigma) > \Qfun((\delta+\mu/2)/\sigma)=A^2$),
the sign of $dM_S/d\delta$ is carried entirely by $\phi_-A-\phi_+B$, so
any stationary point of $M_S$ satisfies
\begin{equation}
 \frac{\phi_-}{\phi_+} = \frac{B}{A}
 \iff
 e^{\mu\delta/\sigma^2} = \sqrt{\frac{1-q_-}{q_+}}
 \label{eq:opt-delta-cond}
\end{equation}
Since $\delta=(\sigma^2/\mu)\log\tau_2$ maps the silence half-width
back to the upper likelihood-ratio threshold, the left-hand side of
\eqref{eq:opt-delta-cond} is exactly $\tau_2$, giving the optimal
silence threshold in closed form:
\begin{equation}
 \tau_2^{\star\,2}
 = \frac{P^{\gamma^S}(u^2|H_2)}{P^{\gamma^S}(u^2|H_1)}
 = \frac{P^{\gamma^S}(u^1|H_1)}{P^{\gamma^S}(u^1|H_2)}
 \label{eq:opt-tau}
\end{equation}
Equation~\eqref{eq:opt-tau} reduces the single-parameter search over
$\delta$ in \eqref{eq:rate-optimization} to evaluating a closed-form
condition instead of a numerical line search.

\section{Numerical Evaluation}
\label{sec:numerical}
For the baseline experiment, use balanced priors and $0$ dB SNR:
$p=1/2$ and $\mu=\sigma^2=1$. Then
$q=\Qfun(0.5)\approx0.3085$ and
\begin{equation}
 C_V\approx0.0793.
\end{equation}
Using the optimal thresholds from \eqref{eq:opt-tau},
$\delta^\star\approx0.6170$, i.e. $\tau_1^\star\approx0.5395$,
$\tau_2^\star\approx1.8534$, giving $q_+\approx0.1320$ and
$q_-\approx0.5466$.  Hence
\begin{equation}
 C_S\approx0.1011,\qquad
 \rho\approx0.5854.
 \label{eq:numerical-silent}
\end{equation}
This example predicts a larger asymptotic exponent while about
$59\%$ of the sensors transmit.  A complete evaluation should compare
finite-$N$ MAP error probabilities, the limits in
\eqref{eq:vanilla-exponent} and \eqref{eq:silent-exponent}, and the
rate--exponent curve obtained by varying $\delta$.

\section{Simulation}
\label{sec:simulation}
Figure~\ref{fig:active-senders-chernoff} plots the simulated normalized
error exponent $J^N_{EE}$ against the number sensors, 
for both the always-transmit and silent policies, alongside
their respective Chernoff-information asymptotes $C_V$ and $C_S$
computed in Section~\ref{sec:numerical}. Both curves decay from a
large-$N$-mismatch regime and converge to their asymptote as the
sensor count grows, confirming \eqref{eq:mixed-exponent} as the
correct $N \to \infty$ limit; the silent policy's curve lies uniformly
above the always-transmit one, consistent with $C_S > C_V$.

\section{Conclusion}
The proposed structure combines the optimal-policy results for large
sensor populations with a rate-aware encoder. Notably, the silent
policy attains a strictly larger error exponent than the
always-transmit policy while sending less data, showing that selective
silence can simultaneously reduce communication and improve detection
performance.
